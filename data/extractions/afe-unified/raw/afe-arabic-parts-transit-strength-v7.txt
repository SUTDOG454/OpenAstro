2. Arabic Parts Engine – Complete Implementation

2.1 General Formula

Core Calculation Method for All Arabic Parts:

1. Identify two significators (planets, house cusps, or special degrees)
2. Calculate arc: Arc = Longitude(Significator B) - Longitude(Significator A)
3. If Longitude(B) < Longitude(A), add 360° to Longitude(B) first
4. Add arc to projection point (usually Ascendant): Sum = Arc + Longitude(Projection Point)
5. Project sum from 0° of projection sign: Part Location = Projection Sign (0°) + Sum

Projection Points:

· Ascendant (most common)
· Beginning of a specific sign
· Degree of a planet

```python
import math
from typing import Dict, List, Optional, Tuple
from enum import Enum

class ChartSect(Enum):
    DIURNAL = "diurnal"  # Sun above horizon (day chart)
    NOCTURNAL = "nocturnal"  # Sun below horizon (night chart)

class ArabicPartsEngine:
    """
    Calculates all Arabic Parts (Lots) for a given chart.
    Supports diurnal/nocturnal variations, multiple projection points, and special cases.
    """

    def __init__(self, positions: Dict[str, float], ascendant: float,
                 house_cusps: Dict[int, float], sun_above_horizon: bool):
        """
        positions: dict of planet -> longitude in degrees
        ascendant: longitude of Ascendant
        house_cusps: dict of house number -> longitude
        sun_above_horizon: True if Sun is above the horizon (day chart)
        """
        self.positions = positions
        self.ascendant = ascendant
        self.house_cusps = house_cusps
        self.sect = ChartSect.DIURNAL if sun_above_horizon else ChartSect.NOCTURNAL
        self.parts = {}

    # ----------------------------------------------------------------------
    # Core Calculation Methods
    # ----------------------------------------------------------------------

    def _measure_arc(self, point_a: float, point_b: float) -> float:
        """
        Calculate arc from A to B (B - A), normalised to 0-360.
        """
        arc = (point_b - point_a) % 360
        return arc

    def _calculate_part(self, a: str, b: str, projection: str) -> float:
        """
        General formula: Part = Projection + (B - A)
        """
        point_a = self._get_position(a)
        point_b = self._get_position(b)
        proj = self._get_position(projection)
        arc = self._measure_arc(point_a, point_b)
        part = (proj + arc) % 360
        return part

    def _calculate_part_diurnal_nocturnal(self, a: str, b: str, projection: str,
                                           diurnal_formula: Tuple[str, str, str],
                                           nocturnal_formula: Tuple[str, str, str]) -> float:
        """
        Calculate a part that has different formulas for diurnal and nocturnal charts.
        """
        if self.sect == ChartSect.DIURNAL:
            return self._calculate_part(*diurnal_formula)
        else:
            return self._calculate_part(*nocturnal_formula)

    def _get_position(self, point: str) -> float:
        """
        Get longitude of a point (planet, house cusp, or special degree).
        """
        if point.startswith("House"):
            house_num = int(point.split("_")[1])
            return self.house_cusps.get(house_num, 0.0)
        elif point.startswith("Sign_"):
            sign_num = int(point.split("_")[1])
            return sign_num * 30
        else:
            return self.positions.get(point, 0.0)

    def _get_sign_and_house(self, longitude: float) -> Tuple[int, int]:
        """
        Get sign number (0-11) and house number (1-12) for a longitude.
        """
        sign = int(longitude // 30) % 12
        # Determine house from cusps
        house = 1
        cusps = sorted(self.house_cusps.items())
        for i, (h, c) in enumerate(cusps):
            next_c = cusps[(i + 1) % len(cusps)][1]
            if h == 1:
                if longitude >= c or longitude < next_c:
                    house = h
                    break
            else:
                if c <= longitude < next_c:
                    house = h
                    break
        return sign, house

    # Specific Part Calculations
    # ----------------------------------------------------------------------

    def calculate_part_of_fortune(self) -> float:
        """
        Pars Fortunae (Part of Fortune)
        Diurnal: Asc + (Moon - Sun)
        Nocturnal: Asc + (Sun - Moon)
        """
        if self.sect == ChartSect.DIURNAL:
            return self._calculate_part("Sun", "Moon", "Ascendant")
        else:
            return self._calculate_part("Moon", "Sun", "Ascendant")

    def calculate_part_of_future(self) -> float:
        """
        Pars Futurorum (Part of Future/Sun)
        Diurnal: Asc + (Sun - Moon)
        Nocturnal: Asc + (Moon - Sun)
        """
        if self.sect == ChartSect.DIURNAL:
            return self._calculate_part("Moon", "Sun", "Ascendant")
        else:
            return self._calculate_part("Sun", "Moon", "Ascendant")

    def calculate_part_of_saturn(self) -> float:
        """
        Pars Saturni (Part of Saturn)
        Diurnal: Asc + (Pars Fortunae - Saturn)
        Nocturnal: Asc + (Saturn - Pars Fortunae)
        """
        fortune = self.calculate_part_of_fortune()
        if self.sect == ChartSect.DIURNAL:
            arc = (fortune - self.positions.get("Saturn", 0)) % 360
            return (self.ascendant + arc) % 360
        else:
            arc = (self.positions.get("Saturn", 0) - fortune) % 360
            return (self.ascendant + arc) % 360

    def calculate_part_of_life(self) -> float:
        """
        Pars Vitae (Part of Life)
        Diurnal: Asc + (Saturn - Jupiter)
        Nocturnal: Asc + (Jupiter - Saturn)
        """
        if self.sect == ChartSect.DIURNAL:
            return self._calculate_part("Jupiter", "Saturn", "Ascendant")
        else:
            return self._calculate_part("Saturn", "Jupiter", "Ascendant")

    def calculate_part_of_father(self) -> float:
        """
        Pars Patris (Part of Father)
        Diurnal: Asc + (Saturn - Sun)
        Nocturnal: Asc + (Sun - Saturn)
        Special: If Saturn is combust, use Jupiter instead
        """
        saturn = self.positions.get("Saturn", 0)
        sun = self.positions.get("Sun", 0)
        # Check if Saturn is combust (within 8° of Sun)
        diff = min(abs(saturn - sun), 360 - abs(saturn - sun))
        if diff < 8:
            # Use Jupiter instead of Saturn
            if self.sect == ChartSect.DIURNAL:
                return self._calculate_part("Jupiter", "Sun", "Ascendant")
            else:
                return self._calculate_part("Sun", "Jupiter", "Ascendant")
        else:
            if self.sect == ChartSect.DIURNAL:
                return self._calculate_part("Sun", "Saturn", "Ascendant")
            else:
                return self._calculate_part("Saturn", "Sun", "Ascendant")

    def calculate_part_of_mother(self) -> float:
        """
        Pars Matrum (Part of Mother)
        Diurnal: Asc + (Moon - Venus)
        Nocturnal: Asc + (Venus - Moon)
        """
        if self.sect == ChartSect.DIURNAL:
            return self._calculate_part("Venus", "Moon", "Ascendant")
        else:
            return self._calculate_part("Moon", "Venus", "Ascendant")

    def calculate_part_of_marriage(self) -> float:
        """
        Pars Coniugii (Part of Marriage, Men)
        Formula: Asc + (Venus - Saturn) (same for both diurnal and nocturnal)
        """
        return self._calculate_part("Saturn", "Venus", "Ascendant")

    def calculate_part_of_death(self) -> float:
        """
        Pars Mortis (Part of Death)
        Formula: Moon → 8th house cusp + Saturn's completed degrees in its sign,
        project from beginning of Saturn's sign
        """
        moon = self.positions.get("Moon", 0)
        eighth_house = self.house_cusps.get(8, 0)
        saturn = self.positions.get("Saturn", 0)
        # Saturn's completed degrees in its sign
        saturn_sign = int(saturn // 30) % 12
        saturn_deg_in_sign = saturn - (saturn_sign * 30)
        arc = (eighth_house - moon) % 360
        # Project from beginning of Saturn's sign
        sign_start = saturn_sign * 30
        part = (sign_start + arc + saturn_deg_in_sign) % 360
        return part

    def calculate_part_of_kingship(self) -> float:
        """
        Pars Regni (Part of Kingship)
        Diurnal: Asc + (Moon - Mars)
        Nocturnal: Asc + (Mars - Moon)
        """

            return self._calculate_part("Mars", "Moon", "Ascendant")

        Pars Frumentum (Part of Grain)

        """
        return self._calculate_part("Sun", "Mars", "Ascendant")
```

```python
            return 5

        if self.exaltations.get(sign) == planet:
            return 4

                    "Earth": ["Taurus", "Virgo", "Capricorn"],
                    "Air": ["Gemini", "Libra", "Aquarius"],
                    "Water": ["Cancer", "Scorpio", "Pisces"]}
        planet_sign = self.get_sign(self.natal.get(planet, 0))
        if sign in elements.get("Fire", []) and planet_sign in elements.get("Fire", []):
            return 3
        if sign in elements.get("Earth", []) and planet_sign in elements.get("Earth", []):
            return 3
        if sign in elements.get("Air", []) and planet_sign in elements.get("Air", []):
            return 3
        if sign in elements.get("Water", []) and planet_sign in elements.get("Water", []):

        """
        lon1 = self.transit.get(transit_planet, 0)
        lon2 = self.natal.get(natal_planet, 0)
        angle = (lon1 - lon2) % 360
        if angle > 180:
            angle = 360 - angle

        aspect_type = self.get_aspect_type(angle)
        orb = min(abs(angle - self._get_aspect_angle(aspect_type)), 180)

        # Essential Dignity Score
        dignity_score = self.get_essential_dignity_score(transit_planet, lon1)

        # Dispositor Analysis
        final_dispositor = self.get_final_dispositor(transit_planet)
        dispositor_score = self.get_essential_dignity_score(final_dispositor,
                                                            self.natal.get(final_dispositor, 0))

        # Accidental Modifiers
        speed_mod = self.get_speed_modifier(transit_planet)
        house_mod = self.get_house_modifier(lon1)
        avashta = self.get_avashta_state(transit_planet, lon1)
        avashta_mod = self.get_avashta_modifier(avashta)
        orb_mod = self.get_orb_modifier(orb)

        # Aspect Modifier
        aspect_mod = self.get_aspect_modifier(aspect_type)

        # Final strength
        strength = (dignity_score + dispositor_score) * speed_mod * house_mod * avashta_mod * orb_mod * aspect_mod

        return {
            "transit_planet": transit_planet,
            "natal_planet": natal_planet,
            "aspect": aspect_type,
            "orb": orb,
            "dignity_score": dignity_score,
            "final_dispositor": final_dispositor,
            "dispositor_score": dispositor_score,
            "speed_modifier": speed_mod,
            "house_modifier": house_mod,
            "avashta": avashta.value,
            "avashta_modifier": avashta_mod,
            "orb_modifier": orb_mod,
            "aspect_modifier": aspect_mod,
            "strength": strength,
            "active": orb <= 8.0
        }

    def _get_aspect_angle(self, aspect_type: str) -> float:
        angle_map = {
            "conjunction": 0,
            "semi-sextile": 30,
            "semi-square": 45,
            "sextile": 60,
            "quintile": 72,
            "square": 90,
            "trine": 120,
            "sesquiquadrate": 135,
            "quincunx": 150,
            "opposition": 180,
            "biquintile": 144
        }
        return angle_map.get(aspect_type, 0)

    def compute_all_transits(self) -> List[Dict]:
        """
        Compute strength for all active transits.
        """
        results = []
        for t_planet in self.transit.keys():
            for n_planet in self.natal.keys():
                result = self.compute_transit_strength(t_planet, n_planet)
                if result["active"]:
                    results.append(result)
        return results
```

```python
from typing import Dict, List, Optional

class HouseAffinityEngine:
    """
    Evaluates planetary house affinities for accidental strength/weakness.
    """

    def __init__(self):
        self.affinities = {
            "Sun": {"strong": [1, 5, 10], "weak": [4, 6, 7, 8, 12]},
            "Moon": {"strong": [2, 4], "weak": [6, 8, 10, 12]},
            "Mercury": {"strong": [1, 3], "weak": [7, 8, 12]},
            "Venus": {"strong": [2, 4, 7], "weak": [6, 8, 10]},
            "Mars": {"strong": [1, 10], "weak": [4, 6, 12]},
            "Jupiter": {"strong": [1, 4, 9], "weak": [6, 7, 8]},
            "Saturn": {"strong": [7, 10, 11], "weak": [1, 4]}
        }

    def get_affinity_bonus(self, planet: str, house: int) -> float:
        """
        Return affinity bonus for a planet in a given house.
        Strong: +1.5, Weak: 0.5, Neutral: 1.0
        """
        if planet in self.affinities:
            if house in self.affinities[planet]["strong"]:
                return 1.5
            elif house in self.affinities[planet]["weak"]:
                return 0.5
        return 1.0

    def get_interpretation(self, planet: str, house: int) -> str:
        """
        Return interpretation for a planet in a house.
        """
        if planet in self.affinities:
            if house in self.affinities[planet]["strong"]:
                return self.affinities[planet]["strong_interpretation"]
            elif house in self.affinities[planet]["weak"]:
                return self.affinities[planet]["weak_interpretation"]
        return "Neutral house placement."
```

```python
    def _is_sun_above_horizon(self, sun_longitude: float, ascendant: float) -> bool:
        # Simplified: Sun above horizon if its longitude is between ASC and DSC
        # In a real implementation, use house position
        desc = (ascendant + 180) % 360
        if ascendant < desc:
            return ascendant < sun_longitude < desc
        else:
            return sun_longitude > ascendant or sun_longitude < desc

    def _get_house(self, longitude: float, house_cusps: Dict[int, float]) -> int:
        cusps = sorted(house_cusps.items())
        for i, (h, c) in enumerate(cusps):
            next_c = cusps[(i + 1) % len(cusps)][1]
            if h == 1:
                if longitude >= c or longitude < next_c:
                    return h
            else:
                if c <= longitude < next_c:
                    return h
        return 1
```
